        async function loadBooks() {
            try {
                const response = await fetch('/api/books');
                const books = await response.json();

                const usersResponse = await fetch('/api/users');
                const users = await usersResponse.json();

                const borrowingsResponse = await fetch('/api/borrowings');
                const borrowings = await borrowingsResponse.json();

                const borrowedBooksElement = document.querySelector(
                    '.stat-card:nth-child(3) .stat-info h3'
                );

                if (borrowedBooksElement) {
                    const activeBorrowings = borrowings.filter(
                        borrowing => borrowing.status === 'Issued'
                    );

                    borrowedBooksElement.textContent = activeBorrowings.length;
                }

                const registeredUsersElement = document.querySelector(
                    '.stat-card:nth-child(2) .stat-info h3'
                );

                if (registeredUsersElement) {
                    registeredUsersElement.textContent = users.length;
                }

                // Total Books
                const totalBooksElement = document.querySelector(
                    '.stat-card:nth-child(1) .stat-info h3'
                );

                if (totalBooksElement) {
                    totalBooksElement.textContent = books.length;
                }

                // Available Books
                const availableBooksElement = document.querySelector(
                    '.stat-card:nth-child(4) .stat-info h3'
                );

                if (availableBooksElement) {
                    const availableBooks = books.reduce(
                        (total, book) => total + book.availableQuantity,
                        0
                    );

                    availableBooksElement.textContent = availableBooks;
                }

                // Recent Books table
                const tableBody = document.querySelector('table tbody');

                if (tableBody) {
                    tableBody.innerHTML = '';

                    books.forEach(book => {
                        const row = document.createElement('tr');

                        const status =
                            book.availableQuantity > 0
                                ? '<span class="status available">Available</span>'
                                : '<span class="status borrowed">Borrowed</span>';

                        row.innerHTML = `
    <td><strong></strong></td>
    <td></td>
    <td></td>
    <td>${status}</td>
    <td>
        <button onclick="editBook(${book.id})"
            style="background:#2563eb;color:white;border:none;padding:7px 10px;border-radius:5px;cursor:pointer;">
            Edit
        </button>

        <button onclick="deleteBook(${book.id})"
            style="background:#dc2626;color:white;border:none;padding:7px 10px;border-radius:5px;cursor:pointer;margin-left:5px;">
            Delete
        </button>
    </td>
`;

                        row.cells[0].querySelector("strong").textContent = book.title;
                        row.cells[1].textContent = book.author;
                        row.cells[2].textContent = book.category || "General";

                        tableBody.appendChild(row);
                    });
                }

            } catch (error) {
                console.error('Error loading books:', error);
            }
        }

        document.addEventListener('DOMContentLoaded', loadBooks);
        async function searchBooks() {
            const keyword = document.getElementById("bookSearch").value.trim();

            if (!keyword) {
                await loadBooks();
                return;
            }

            try {
                const response = await fetch(
                    `/api/books/search?keyword=${encodeURIComponent(keyword)}`
                );

                if (!response.ok) {
                    throw new Error("Unable to search books.");
                }

                const books = await response.json();
                const tableBody = document.querySelector("table tbody");

                tableBody.innerHTML = "";

                if (books.length === 0) {
                    tableBody.innerHTML = `
                <tr>
                    <td colspan="4" style="text-align:center;">
                        No books found.
                    </td>
                </tr>
            `;
                    return;
                }

                books.forEach(book => {
                    const row = document.createElement("tr");

                    const status = book.availableQuantity > 0
                        ? '<span class="badge available">Available</span>'
                        : '<span class="badge borrowed">Unavailable</span>';

                    row.innerHTML = `
    <td><strong></strong></td>
    <td></td>
    <td></td>
    <td>${status}</td>
    <td>
        <button onclick="editBook(${book.id})"
            style="background:#2563eb;color:white;border:none;padding:7px 10px;border-radius:5px;cursor:pointer;">
            Edit
        </button>

        <button onclick="deleteBook(${book.id})"
            style="background:#dc2626;color:white;border:none;padding:7px 10px;border-radius:5px;cursor:pointer;margin-left:5px;">

        async function issueBook() {
            const userId = document.getElementById("issueUser").value;
            const bookId = document.getElementById("issueBook").value;
            const dueDate = document.getElementById("dueDate").value;

            if (!userId || !bookId || !dueDate) {
                alert("Please select a user, a book, and a due date.");
                return;
            }

            if (dueDate < new Date().toISOString().split("T")[0]) {
                alert("Due date cannot be in the past.");
                return;
            }

            try {
                const response = await fetch(
                    `/api/borrowings/issue?userId=${encodeURIComponent(userId)}&bookId=${encodeURIComponent(bookId)}&dueDate=${encodeURIComponent(dueDate)}`,
                    {
                        method: "POST"
                    }
                );

                if (!response.ok) {
                    const error = await response.text();
                    throw new Error(error || "Failed to issue book.");
                }

                alert("Book issued successfully!");

                closeIssueBookModal();
                document.getElementById("issueUser").value = "";
                document.getElementById("issueBook").value = "";
                document.getElementById("dueDate").value = "";

                await loadBooks();

            } catch (error) {
                console.error("Issue book error:", error);
                alert("Book issue failed: " + error.message);
            }
        }
        async function openReturnBookModal() {
            document.getElementById("returnBookModal").style.display = "flex";

            const select = document.getElementById("returnBorrowing");
            select.innerHTML = '<option value="">Loading issued books...</option>';

            try {
                const [borrowingsResponse, booksResponse, usersResponse] =
                    await Promise.all([
                        fetch("/api/borrowings"),
                        fetch("/api/books"),
                        fetch("/api/users")
                    ]);

                if (!borrowingsResponse.ok || !booksResponse.ok || !usersResponse.ok) {
                    throw new Error("Unable to load library records.");
                }

                const borrowings = await borrowingsResponse.json();
                const books = await booksResponse.json();
                const users = await usersResponse.json();

                const issuedBooks = borrowings.filter(
                    borrowing => borrowing.status === "Issued"
                );

                select.innerHTML = '<option value="">Select an issued book</option>';

                issuedBooks.forEach(borrowing => {
                    const book = books.find(item => item.id === borrowing.bookId);
                    const user = users.find(item => item.id === borrowing.userId);

                    const option = document.createElement("option");
                    option.value = borrowing.id;
                    option.textContent =
                        `${book ? book.title : "Unknown Book"} — ${user ? user.name : "Unknown User"}`;

                    select.appendChild(option);
                });

                if (issuedBooks.length === 0) {
                    select.innerHTML = '<option value="">No books currently issued</option>';
                }
            } catch (error) {
                console.error(error);
                select.innerHTML = '<option value="">Unable to load issued books</option>';
                alert(error.message);
            }
        }

        async function addNewBook() {
            const title = document.getElementById("newBookTitle").value.trim();
            const author = document.getElementById("newBookAuthor").value.trim();
            const category = document.getElementById("newBookCategory").value.trim();
            const isbn = document.getElementById("newBookISBN").value.trim();
            const quantity = Number(document.getElementById("newBookQuantity").value);

            if (!title || !author) {
                alert("Please enter book title and author name.");
                return;

            if (!Number.isInteger(quantity) || quantity < 1) {
                alert("Quantity must be at least 1.");
                return;
            }

            const book = {
                title: title,
                author: author,
                category: category || null,
                isbn: isbn || null,
                quantity: quantity,
                availableQuantity: quantity
            };

            try {
                const response = await fetch("/api/books", {
                    method: "POST",
                    headers: {
                        "Content-Type": "application/json"
                    },
                    body: JSON.stringify(book)
                });

                if (!response.ok) {
                    const message = await response.text();
                    throw new Error(message || "Unable to save book.");
                }

                alert("Book added successfully!");

                closeAddBookModal();

                document.getElementById("newBookTitle").value = "";
                document.getElementById("newBookAuthor").value = "";
                document.getElementById("newBookCategory").value = "";
                document.getElementById("newBookISBN").value = "";
                document.getElementById("newBookQuantity").value = "1";

                await loadBooks();

            } catch (error) {
                console.error("Add book error:", error);
                alert("Book add failed: " + error.message);
            }
        }
        async function editBook(id) {
            try {
                const response = await fetch(`/api/books`);

                if (!response.ok) {
                    throw new Error("Unable to load books.");
                }

                const books = await response.json();
                const book = books.find(item => item.id === id);

                if (!book) {
                    alert("Book not found.");
                    return;
                }

                const title = prompt("Book Title:", book.title);
                if (title === null) return;

                const author = prompt("Author Name:", book.author);
                if (author === null) return;

                const category = prompt("Category:", book.category || "");
                if (category === null) return;

                const isbn = prompt("ISBN:", book.isbn || "");
                if (isbn === null) return;

                const quantityInput = prompt("Total Quantity:", book.quantity);
                if (quantityInput === null) return;

                const quantity = Number(quantityInput);

                if (!title.trim() || !author.trim()) {
                    alert("Title and author are required.");
                    return;
                }

                if (!Number.isInteger(quantity) || quantity < 1) {
                    alert("Quantity must be at least 1.");
                    return;
                }

                const updateResponse = await fetch(`/api/books/${id}`, {
                    method: "PUT",
                    headers: {
                        "Content-Type": "application/json"
                    },
                    body: JSON.stringify({
                        title: title.trim(),
                        author: author.trim(),
                        category: category.trim() || null,
                        isbn: isbn.trim() || null,
                        quantity: quantity
                    })
                });

                if (!updateResponse.ok) {
                    const message = await updateResponse.text();
                    throw new Error(message || "Book update failed.");
                }

                alert("Book updated successfully!");
                await loadBooks();

            } catch (error) {
                console.error("Edit book error:", error);
                alert("Edit failed: " + error.message);
            }
        }

        async function deleteBook(id) {

                alert("Book deleted successfully!");

                console.error("Delete book error:", error);
                alert("Delete failed: " + error.message);
            }
        }

        // Load and search all books
        async function searchAllBooks() {
            const keyword = document.getElementById("allBooksSearch").value.trim();

            try {
                let url = "/api/books";

                if (keyword) {
                    url += `/search?keyword=${encodeURIComponent(keyword)}`;
                }

                const response = await fetch(url);

                if (!response.ok) {
                    throw new Error("Unable to load books.");
                }

                const books = await response.json();
                const tableBody = document.querySelector("#allBooksTable tbody");

                tableBody.innerHTML = "";

                if (books.length === 0) {
                    tableBody.innerHTML = `
                <tr>
                    <td colspan="6" style="text-align:center;">
                        No books found.
                    </td>
                </tr>
            `;
                    return;
                }

                books.forEach(book => {
                    const row = document.createElement("tr");

                    const status = book.availableQuantity > 0
                        ? "Available"
                        : "Unavailable";

                    row.innerHTML = `
                <td></td>
                <td></td>
                <td></td>
                <td></td>
                <td></td>
                <td>
                    <button onclick="editBook(${book.id})"
                        style="background:#2563eb;color:white;border:none;padding:7px 10px;border-radius:5px;cursor:pointer;">
                        Edit
                    </button>

                    <button onclick="deleteBook(${book.id})"
                        style="background:#dc2626;color:white;border:none;padding:7px 10px;border-radius:5px;cursor:pointer;margin-left:5px;">
                        Delete
                    </button>
                </td>
            `;

                    row.cells[0].textContent = book.title;
                    row.cells[1].textContent = book.author;
                    row.cells[2].textContent = book.category || "General";
                    row.cells[3].textContent = book.quantity;
                    row.cells[4].textContent =
                        `${book.availableQuantity} (${status})`;

                    tableBody.appendChild(row);
                });

            } catch (error) {
                console.error("Books page error:", error);
                alert("Unable to load books: " + error.message);
            }
        }
